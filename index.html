@router.message(PromoState.waiting_for_code, F.text, ~F.text.startswith("/")) async def process_promo(message: Message, state: FSMContext): code = message.text.strip().upper() await state.clear() async with async_session() as session: promo = await session.scalar(select(PromoCode).where(PromoCode.code == code)) if not promo: return await message.answer("❌ Такого промокода не существует.") if promo.uses_left <= 0: return await message.answer("❌ Этот промокод больше не работает.") already_used = await session.scalar( select(UserPromo).where(UserPromo.user_id == message.from_user.id, UserPromo.promo_id == promo.id) ) if already_used: return await message.answer("❌ Ты уже активировал этот промокод.") promo.uses_left -= 1 new_use = UserPromo(user_id=message.from_user.id, promo_id=promo.id) session.add(new_use) await session.commit() await message.answer( f"✅ Промокод {code} активирован!\n\n" f"Твоя скидка: {promo.discount}%.\n" f"Покажи это сообщение менеджеру (@Wogoro)." )
